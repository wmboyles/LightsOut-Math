    \author{William Boyles}

    \begin{abstract}
    	\textit{Lights Out} is a game on a simple graph $G = (V,E)$ where clicking a vertex toggles the on/off state of it and its neighbors.
	    Identifying subsets of $V$ with their indicator vectors over $\F_2$, let $\Phi_G: \F_2^V \rightarrow \F_2^V$ be the function which maps the set of vertices clicked to the set of vertices that change state.
	    For the $n \times n$ grid, define $d(n) = \dim{\ker{\Phi_G}}$.
	    We recover the known fact that $d(n)$ is always even and prove infinitely many even values are also impossible.
	    We provide formulas and bounds for $d(n)$, including a complete characterization when $n = 2^{a}p^{b} - 1$ for $a, b \geq 0$ and $p$ not a Wieferich prime.
	\end{abstract}

	\section{Fibonacci Polynomials}
	We begin by introducing Fibonacci polynomials.
	We first develop some basic properties of these polynomials that will be useful in relating them to the Lights Out problem.
	
	\begin{definition}[Fibonacci Polynomial]
		Let $F_n(x)$ be the polynomial in the ring $\F_2[x]$ defined recursively by
		\begin{equation*}
			F_n(x) = \begin{cases}
				0 & n=0 \\
				1 & n=1 \\
				xF_{n-1}(x) + F_{n-2}(x) & \text{otherwise}
			\end{cases}.
		\end{equation*}
	\end{definition}
	These polynomials are called "Fibonacci" because $F_n(1)$ is the $n$th Fibonacci number when evaluated over $\Z[x]$.
	Evaluated over $\F_2$, $F_n(1)$ is the $n$th Fibonacci number modulo 2.
	
	In $\F_2[x]$, the coefficients of $F_n(x)$ are either 1 or 0, so we can visualize them by coloring squares black or white to represent the coefficients.
	For example,
	\begin{equation*}
		F_6(x) = 1x^5 + 0x^4 + 0x^3 + 0x^2 + 1x + 0 = \blacksquare \square \square \square \blacksquare \square.
	\end{equation*}

	Plotting $F_1(x)$ through $F_{128}(x)$ in the same way, aligning terms of the same degree, we see a Sierpinski triangle.

	\begin{figure}[H]
		\centering
		\includegraphics[width=.8\textwidth]{./chebyshev1_right_128.png}	
		\caption{$F_1(x)$ through $F_{128}(x)$, right-aligned.}
	\end{figure}
	
	The Sierpinski arises from the parity of the binomial coefficients in Pascal's triangle, giving the formula
	\begin{equation*}
		F_n(x) = \sum_{i=0}^{n-1}{\binom{n+i}{2i+1}x^i \mod 2}.
	\end{equation*}

	Sutner also showed the above construction and formula, along with the following relationship between rectangular grids and Fibonacci Polynomials \cite[Theorem 5.1]{Sutner00sigma-automataand}.
	\begin{lemma}[Sutner]\label{FibonacciPolynomialGCD}
		Let $G = (V,E)$ be an $n \times m$ grid graph.
		Then
		\begin{equation*}
			\dim{\ker{\Phi_G}}
			= \deg{\left(\gcd{\left(F_{n+1}(x), F_{m+1}(x+1)\right)}\right)}
			= \deg{\left(\gcd{\left(F_{n+1}(x+1), F_{m+1}(x)\right)}\right)}.
		\end{equation*}
	\end{lemma}
	
	In particular, this lemma allows us to define $d(n)$ in terms of Fibonacci polynomials.
	\begin{equation*}
		d(n) = \deg{\left(\gcd{\left(F_{n+1}(x), F_{n+1}(x+1)\right)}\right)}.
	\end{equation*}
	
	Sutner and others also established several divisibility properties of Fibonacci polynomials \cite[Proposition 2.2]{HUNZIKER2004465}\cite[Proposition 2.1]{Sutner00sigma-automataand}\cite[Section 7.1.3, Exercises 190--194]{Knuth_AOCP4A}.
	\begin{lemma}[Hunziker, Machiavelo, and Park]\label{FibonacciDivisibility}
		A polynomial $\tau(x)$ in $\F_2[x]$ divides both $F_n(x)$ and $F_m(x)$ if and only if it divides $F_{\gcd(m,n)}(x)$.
		In particular,
		\begin{equation*}
			\gcd{\left(F_m(x),F_n(x)\right)} = F_{\gcd(m,n)}(x).
		\end{equation*}
	\end{lemma}
	
	\begin{corollary}\label{Threeven}
		The following are true:
		\begin{enumerate}[label=(\roman*)]
			\item $x$ divides $F_n(x)$ if and only if $n \equiv 0 \mod 2$.
			\item $x+1$ divides $F_n(x+1)$ if and only if $n \equiv 0 \mod 2$.
			\item $x+1$ divides $F_n(x)$ if and only if $n \equiv 0 \mod 3$.
			\item $x$ divides $F_n(x+1)$ if and only if $n \equiv 0 \mod 3$.
		\end{enumerate}
	\end{corollary}
	\begin{proof}
		Notice,
		\begin{enumerate}[label=(\roman*)]
			\item We find that $F_2(x) = x$ is the smallest Fibonacci polynomial divisible by $x$, so we apply Lemma \ref{FibonacciDivisibility} to get the desired result.
			\item Follows from (i) by substituting $x+1$ for $x$.
			\item We find that $F_3(x) = (x+1)^2$ is the smallest Fibonacci polynomial divisible by $x+1$, so we apply Lemma \ref{FibonacciDivisibility} to get the desired result.
			\item Follows from (iii) by substituting $x+1$ for $x$.
		\end{enumerate}
	\end{proof}
	
	Hunziker, Machiavelo, and Park give the following characterization of the square-free factors of Fibonacci polynomials \cite[Proposition 2.11]{HUNZIKER2004465}.
	\begin{lemma}[Hunziker, Machiavelo, and Park]\label{SquareSquareFree}
		If $n$ is odd, then $F_n(x)$ is the square of a square-free polynomial.
	\end{lemma}

	\begin{lemma}\label{translationInvarianceField}
		Let $P \in \F_2[x]$ such that $P(x) = P(x+1)$.
		Then $P \in \F_2[x^2 + x]$.
	\end{lemma}
	\begin{proof}
		Let $y = x(x+1)$.
		Every polynomial in $\F_2[x]$ can be written uniquely in the following form
		\begin{equation*}
			P(x) = A(y) + xB(y) \qquad A,B \in \F_2[y].
		\end{equation*}
		Since $P$ is unchanged by translation,
		\begin{align*}
			P(x+1) &= P(x) \\
			A(y) + (x+1)B(y) &= A(y) + xB(y) \\
			B(y) &= 0.
		\end{align*}
		Thus, $P(x) = A(y)$ and is thus in $\F_2[y] = \F_2[x^2 + x]$, as desired.
	\end{proof}

	We can now use these properties to prove several facts about $d(n)$.
	\begin{lemma}\label{DEven}
		Let $d(n) = \dim{\ker{\Phi_G}}$ for $G$ an $n \times n$ grid graph.
		Then $d(n)$ is always even.
	\end{lemma}
	\begin{proof}
		Let
		\begin{equation*}
			G_n(x) = \gcd{\left(F_{n}(x),F_{n}(x+1)\right)}.
		\end{equation*}
		By Lemma \ref{FibonacciPolynomialGCD}, we know that $d(n) = \deg{G_{n+1}(x)}$ and $G_n(x) = G_n(x+1)$.
		Since $G_n(x)$ is unchanged by translation, Lemma \ref{translationInvarianceField} tells us there exists $H_n \in \F_2[x]$ such that
		\begin{equation*}
			G_n(x) = H_n(x^2 + x).
		\end{equation*}
		So,
		\begin{equation*}
			d(n) = \deg{G_{n+1}(x)} = 2\deg{H_{n+1}(x)}
		\end{equation*}
		and thus $d(n)$ is even.
	\end{proof}
	Sutner observed the above lemma computationally \cite{Sutner1989}.

	\begin{corollary}\label{NEvenDDivisible4}
		If $n$ is even, then $d(n)$ is divisible by 4.
	\end{corollary}
	\begin{proof}
		If $n$ is even, then $n+1$ is odd.
		Then by Lemma \ref{SquareSquareFree}, $F_{n+1}(x)$ and $F_{n+1}(x+1)$ are squares of square-free polynomials $\tau(x)$ and $\tau(x+1)$.
		\begin{align*}
			G_{n+1}(x) &= \gcd{\left(F_{n+1}(x),F_{n+1}(x+1)\right)} \\
			&= \gcd{\left(\tau_{n+1}(x)^2, \tau_{n+1}(x+1)^2\right)} \\
			&= \gcd{\left(\tau_{n+1}(x), \tau_{n+1}(x+1)\right)}^2 \\
			\sqrt{G_{n+1}(x)} &= \gcd{\left(\tau_{n+1}(x), \tau_{n+1}(x+1)\right)}
		\end{align*}
		Since translation by 1 exchanges the arguments of the GCD, $\sqrt{G_{n+1}(x)} = \sqrt{G_{n+1}(x+1)}$.
		So, by Lemma \ref{DEven}, there exists $H_{n+1} \in \F_2[x]$ such that
		\begin{align*}
			\sqrt{G_{n+1}(x)} &= H_{n+1}(x^2+x) \\
			G_{n+1}(x) &= H_{n+1}(x^2+x)^2.
		\end{align*}
		So,
		\begin{equation*}
			d(n) = \deg G_{n+1}(x) = 4\deg{H_{n+1}(x)},
		\end{equation*}
		as desired.
	\end{proof}
	Note that the implication does not hold in the other direction: $d(n)$ may also be divisible by 4 when $n$ is odd.
	For example, $d(9) = 8$.

	\begin{lemma}\label{FibonacciFormulaT}
		Let $x = t + t^{-1}$.
		Then for $t \neq 0$,
		\begin{equation*}
			xF_n(x) = t^{n} + t^{-n}.
		\end{equation*}
	\end{lemma}
	\begin{proof}
		Define $G_n(t) = t^{n+1} + t^{-(n+1)}$.
		Then
		\begin{align*}
			xG_{n-1}(t) + G_{n-2}(t) &= (t+t^{-1})(t^{n}+t^{-n}) + (t^{n-1}+t^{-(n-1)}) \\
			&= t^{n+1} + t^{-(n-1)} + t^{n-1} + t^{-(n+1)} + t^{n-1} + t^{-(n-1)} \\
			&= t^{n+1} + t^{-(n+1)} \\
			&= G_n.
		\end{align*}
		Thus we see that $G_n(t)$ satisfies the same recurrence relation as $F_n(x)$.
		$xF_n(x)$ also satisfies the same recurrence relation.
		\begin{align*}
			xF_n(x) &= x(xF_{n-1}(x)+F_{n-2}(x)) \\
			&= x(xF_{n-1}(x)) + xF_{n-2}(x).
		\end{align*}
		Now we compute $xF_1(x)$ and $xF_2(x)$ in terms of $t$ and $G$:
		\begin{align*}
			xF_1(x) &= x \\
			&= t + t^{-1} \\
			&= G_0(t),
		\end{align*}
		and
		\begin{align*}
			xF_2(x) &= x^2 \\
			&= (t+t^{-1})^2 \\
			&= t^2 + t^{-2} \\
			&= G_1(t).
		\end{align*}
		Since the first two terms match and they satisfy the same recurrence relation then we've shown by induction that for $t \neq 0$,
		\begin{equation*}
			xF_n(x) = G_{n-1}(t) = t^{n} + t^{-n}.
		\end{equation*}
	\end{proof}
	This was also proven by Hunziker, Machiavelo, and Park \cite[Proposition 2.3]{HUNZIKER2004465}.
	
	The following identity is a specialization of a result of Hunziker, Machiavelo, and Park \cite[Lemma 2.6]{HUNZIKER2004465}.
	\begin{lemma}[Hunziker, Machiavelo, and Park]\label{Fibonacci2k}
		Let $n = 2^k b$ where $k \geq 0$ and $b$ is odd.
		Then
		\begin{equation*}
			F_n(x) = x^{2^k-1}F_b(x)^{2^k}.
		\end{equation*}
	\end{lemma}
	
	\begin{lemma}\label{RootParameterization}
		The distinct roots of $F_n(x)$ in $\overline{\F_2}$ are $\{\zeta + \zeta^{-1} : \zeta^{n} = 1, \zeta \neq 1\}$ together with 0 when $n$ is even.
	\end{lemma}
	\begin{proof}
		Notice that $F_n(0) = F_{n-2}(0)$, so
		\begin{equation*}
			F_n(0) = \begin{cases}
				1 & n \text{ odd} \\
				0 & n \text{ even}
			\end{cases}.
		\end{equation*}
		Thus 0 is a root of $F_n(x)$ whenever $n$ is even.
		
		Let $\alpha$ be a non-zero root of $F_n$.
		Choose $\zeta$ satisfying
		\begin{equation*}
			\zeta^2 + \alpha\zeta + 1 = 0.
		\end{equation*}
		Notice that $\zeta$ cannot be 0.
		Dividing by $\zeta$,
		\begin{align*}
			\zeta + \alpha + \zeta^{-1} &= 0 \\
			\alpha &= \zeta + \zeta^{-1}.
		\end{align*}
		By, Lemma \ref{FibonacciFormulaT},
		\begin{align*}
			0 &= \alpha F_n(\alpha) \\
			&= \zeta^{n} + \zeta^{-n} \\
		\end{align*}
		Since a field has no nonzero nilpotents, we may multiply by $\zeta^n$.
		\begin{align*}
			0 &= \zeta^{2n} + 1 \\
			&= \left(\zeta^{n}+1\right)^2,
		\end{align*}
		and
		\begin{align*}
			0 &= \zeta^{n} + 1 \\
			\zeta^{n} &= 1.
		\end{align*}
		Thus the order of $\zeta$ divides $n$.
		Notice too that $\zeta \neq 1$; otherwise, $\alpha = 1 + 1^{-1} = 0$, which we already showed as not possible.
		
		Conversely, if $\zeta^{n} = 1$ and $\zeta \neq 1$, then let $\alpha = \zeta + \zeta^{-1}$.
		Then we see $\alpha \neq 0$ and
		\begin{equation*}
			\alpha F_n(\alpha) = \zeta^{n} + \zeta^{-n} = 1 + 1^{-1} = 0.
		\end{equation*}
		Thus, $\alpha$ is a root of $F_n$.
	\end{proof}

	This parameterization gives a convenient description of the roots of Fibonacci polynomials in terms of roots of unity.

	\section{Even Dominating Sets}
	We identify a subset $S \subseteq V$ with its indicator vector $\mathbf{1}_{S} \in \F_2^V$.
	Under this identification, symmetric difference of subsets corresponds to addition in $\F_2^V$.
	The map $\Phi_G$ is linear since the state change of each vertex is determined by the parity of the clicks on its closed neighborhood.
	Consequently, two configurations $V_1, V_2 \subseteq V$ have the same image under $\Phi_G$ if and only if $V_1 \triangle V_2 \in \ker{\Phi_G}$.
	Thus, configurations producing the same state change form cosets of $\ker{\Phi_G}$.
	In particular, whenever an initial configuration is solvable, the set of all solutions is a coset of $\ker{\Phi_G}$, and hence has $2^d$ elements, where $d = \dim{\ker{\Phi_G}}$.
	$\ker{\Phi_G}$ also has a natural interpretation in terms of even dominating sets.
	
	\begin{definition}
		An \textit{even dominating set} of a graph $G = (V,E)$ is a subset $V' \subseteq V$ such that the closed neighborhood $N[v]$ of every vertex contains an even number of vertices from $V'$. 
	\end{definition}
	Odd dominating sets can be analogously defined.
	Sutner specifically studied odd dominating sets as the solution to \textit{Lights Out} when all lights are initially on.
	He showed that such an odd dominating set always exists for any finite graph \cite{Sutner1989}.
	
	\begin{lemma}
		$V' \in \ker{\Phi_G}$ if and only if $V'$ is an even dominating set of $G$.
	\end{lemma}
	\begin{proof}
		Let $V' \in \ker{\Phi_G}$.
		Then every vertex is toggled an even number of times when the vertices in $V'$ are clicked.
		Since each vertex is toggled exactly when the vertices in its closed neighborhood are clicked, for every $v \in V$,
		\begin{equation*}
			\abs{N[v] \cap V'} \text{ is even}.
		\end{equation*}
		Thus, $V'$ is an even dominating set of $G$.
		The converse follows by reversing the same argument.
	\end{proof}

	We may now use our graph-theoretic understanding of $\ker{\Phi_G}$ as even dominating sets to construct even dominating sets on larger grids by tiling those from smaller grids.
	\begin{theorem}\label{nk-1>n-1}
		For all $n,k \in \N$,
		\begin{equation*}
			d(nk-1) \geq d(n-1).
		\end{equation*}
	\end{theorem}
	\begin{proof}
		Let $G_L$ be an $nk-1 \times nk-1$ grid and $G_\ell$ be an $n-1 \times n-1$ grid.
		Since $nk-1 = k(n-1) + (k-1)$, an $G$ may be broken into a $k \times k$ grid of $n-1 \times n-1$ grids, each separated by a horizontal or vertical strip of size 1.
		
		Let $Q$ be an even dominating set of $G_\ell$.
		Let $R$ be $Q$ tiled $k$ times horizontally and vertically, with size 1 separators between each tile, onto $G_L$ where each tile is the reflection across its neighboring separators, as shown below.

		Consider a vertex $v$ in $G_L$.
		There are 3 cases.
		\begin{enumerate}
			\item
				$v$ is within a tile.
				Then $\abs{N[v] \cap R}$ is even because $Q$ is an even dominating set.
				Vertices next to tile separators are unaffected by the tiling because the separator vertices are not included in $R$ and thus don't contribute to the parity of $N[v] \cap R$.
				Thus, $\abs{N[v] \cap R}$ has the same parity as the corresponding closed neighborhood in $Q$, which is even.
			\item
				$v$ is on a separator between two tiles.
				Then only two elements in $N[v]$, the vertices within a tile, are also in $R$.
				Because neighboring tiles are reflections of each other across separators, these two vertices are either both in $R$ or both not in $R$.
				In either case, $\abs{N[v] \cap R}$ is even.
			\item
				$v$ is in the intersection of two separators.
				Then all of $N[v]$ is in a separator and thus not in $R$, so $\abs{N[v] \cap R} = 0$, which is even.
		\end{enumerate}
		Since for every vertex $v$ of $G_L$, $\abs{N[v] \cap R}$ is even, so $R$ is an even dominating set of $G_L$.
		Thus, for every even dominating set of $G_\ell$, there exists an even dominating set of $G_L$ obtained through tiling.
		This tiling function is also injective because we can recover $Q$ by restricting $R$ to just the top left tile.
		Thus,
		\begin{equation*}
			d(nk-1) \geq d(n-1),
		\end{equation*}
		as desired.
	\end{proof}
	Sutner shows a similar result \cite{Sutner1988_2}.

	The same inequality also follows from the Fibonacci polynomial characterization.
	\begin{proof}
		Since $n$ divides $nk$, Lemma \ref{FibonacciDivisibility} tells us $F_{n}(x)$ divides $F_{nk}(x)$.
		Substituting $x+1$, we also have $F_{n}(x+1)$ divides $F_{nk}(x+1)$.
		Thus, the first GCD divides the second:
		\begin{equation*}
			\gcd{\left(F_n(x),F_n(x+1)\right)} \mid \gcd{\left(F_{nk}(x),F_{nk}(x+1)\right)}.
		\end{equation*}
		So we may write
		\begin{equation*}
			\gcd{\left(F_{nk}(x),F_{nk}(x+1)\right)} = 	\gcd{\left(F_n(x),F_n(x+1)\right)} Q(x),
		\end{equation*}
		for some $Q(x) \in \F_2[x]$.
		Taking the degrees, we obtain $d(nk-1) = d(n-1) + \deg{Q(x)}$, and the desired inequality follows.
	\end{proof}
	Note that although the smaller GCD polynomial divides the larger, it's not the case that $d(n-1)$ divides $d(nk-1)$.
	For example, if $n=5$ and $k=6$, then $d(n-1) = d(4) = 4$, which does not divide $d(nk-1) = d(29) = 10$.

	\section{Number Theory}
	We now use results from number theory to study the GCDs of Fibonacci polynomials and, consequently, the values of $d(n)$.
	Many of the results in this section first appeared in the author's earlier works \cite{boylesMostClicks, boylesResolutionSutner}.
	We restate several proofs here for completeness and to correct minor errors in those earlier versions.
	
	\begin{lemma}[Ore]\label{OreGcd}
		Let $a$, $b$, $c$, and $d$ be integers.
		Let $(a,b)$ denote $\gcd{(a,b)}$.
		Then
		\begin{equation*}
			(ab,cd) = (a,c)(b,d)\left(\frac{a}{(a,c)},\frac{d}{(b,d)}\right)\left(\frac{c}{(a,c)},\frac{b}{(b,d)}\right).
		\end{equation*}
	\end{lemma}
	We will also use the analogous identity in $\F_2[x]$.
	Since $\F_2[x]$ is a Euclidean domain, the same proof applies with divisibility and GCDs interpreted in $\F_2[x]$ \cite[Section 5-4, Problem 2, p.~109]{ore_number_theory}.
	
	We can now combine the preceding polynomial identity to obtain a recurrence for $d(n)$.
	\begin{theorem}\label{2nPlus1}
		For all $n \in \N$,
		\begin{equation*}
			d(2n+1) = 2d(n) + \delta_n, \qquad
			\delta_n = \begin{cases}
				2 & 3 \mid n+1 \\
				0 & \text{otherwise}
			\end{cases}.
		\end{equation*}
	\end{theorem}
	\begin{proof}
		Let $(a,b)$ denote $\gcd{(a,b)}$.
		Applying Lemmas \ref{Fibonacci2k} and \ref{OreGcd},
		\begin{align*}
			d(2n+1) &= \deg{\left(F_{2n+2}(x),F_{2n+2}(x+1)\right)} \\
			&= \deg{\left(xF_{n+1}(x)^2,(x+1)F_{n+1}(x+1)^2\right)} \\
			&= \deg (x,x+1)\left(F_{n+1}(x)^2,F_{n+1}(x+1)^2\right) \\
			&\qquad\times \left(\frac{x+1}{(x,x+1)},\frac{F_{n+1}(x)^2}{\left(F_{n+1}(x)^2,F_{n+1}(x+1)^2\right)}\right)\left(\frac{x}{(x,x+1)},\frac{F_{n+1}(x+1)^2}{\left(F_{n+1}(x)^2,F_{n+1}(x+1)^2\right)}\right) \\
			&= \deg{\left(F_{n+1}(x),F_{n+1}(x+1)\right)^2\left(x+1,\frac{F_{n+1}(x)^2}{\left(F_{n+1}(x),F_{n+1}(x+1)\right)^2}\right)\left(x,\frac{F_{n+1}(x+1)^2}{\left(F_{n+1}(x),F_{n+1}(x+1)\right)^2}\right)} \\
			&= 2d(n) + 2\deg{\left(x,\frac{F_{n+1}(x+1)^2}{\left(F_{n+1}(x),F_{n+1}(x+1)\right)^2}\right)}
		\end{align*}
		Thus, we have that
		\begin{equation*}
			\delta_n = 2\deg{\left(x,\frac{F_{n+1}(x+1)^2}{\left(F_{n+1}(x),F_{n+1}(x+1)\right)^2}\right)}.
		\end{equation*}
		So, $\delta_n$ is 2 exactly when $x+1$ occurs with greater multiplicity in $F_{n+1}(x)$ than in $F_{n+1}(x+1)$.
		When $n+1$ is not divisible by 3, Corollary \ref{Threeven} tells us that $F_{n+1}(x)$ is not divisible by $x+1$, so $\delta_n = 0$.
		When $n+1$ is divisible by 3, let $n+1 = b 2^{k}$ for $b,k \geq 0$ and $b$ odd.
		Then $b$ must be divisible by 3.
		Applying Lemma \ref{Fibonacci2k},
		\begin{equation*}
			F_{n+1}(x) = x^{2^{k}-1}{F_{b}(x)}^{2^k} \qquad F_{n+1}(x+1) = (x+1)^{2^{k}-1}{F_{b}(x+1)}^{2^k}.
		\end{equation*}
		Since $b$ is odd, Lemma \ref{SquareSquareFree} tells us $F_b(x)$ is the square of a square-free polynomial.
		Since $b$ is a multiple of 3, Corollary \ref{Threeven} tells us that $x+1$ divides $F_{b}(x)$, but $x+1$ does not divide $F_b(x+1)$.
		So,
		\begin{equation*}
			F_{n+1}(x) = x^{2^{k}-1}(x+1)^{2^{k+1}}g(x)^{2^{k+1}} \qquad F_{n+1}(x+1) = x^{2^{k+1}}(x+1)^{2^{k}-1}g(x+1)^{2^{k+1}},
		\end{equation*}
		for some $g(x) \in \F_2[x]$, where $g(x)$ and $g(x+1)$ are both square-free and divisible by neither $x$ nor $x+1$.
		So, we see that $F_{n+1}(x)$ can be divided without remainder by $x+1$ more times than $F_{n+1}(x+1)$, so $\delta_n = 2$.
	\end{proof}
	This result was first proven by Boyles \cite{boylesResolutionSutner}, resolving a conjecture of Sutner \cite{Sutner1989}.
	
	Repeated application of Theorem \ref{2nPlus1} gives the following explicit formula for $d(n)$.
	\begin{corollary}\label{2nPlus1Explicit}
		Write $n$ as $n+1 = 2^s b$ for $s \geq 0$ and $b$ odd.
		Then
		\begin{equation*}
			d(n) = 2^s d(b-1) + \begin{cases}
				2(2^s - 1) & 3 \mid b \\
				0 & \text{otherwise}
			\end{cases}.
		\end{equation*}
	\end{corollary}
	\begin{proof}
		We proceed by induction on $s$.
		For $s=0$, we have $d(n) = d(b-1)$, which is the desired formula.
		Now suppose the result holds for some $s \geq 0$.
		For $n+1 = 2^{s+1}b$, Theorem \ref{2nPlus1} gives
		\begin{align*}
			d(n) &= d\left(2\left(2^s b - 1\right) + 1\right) \\
			&= 2d\left(2^s b - 1\right) + \begin{cases}
				2 & 3 \mid 2^s b \\
				0 & \text{otherwise}
			\end{cases}.
		\end{align*}
		Since $3 \mid 2^s b$ if and only if $3 \mid b$, the induction hypothesis gives
		\begin{align*}
			d(n) &= 2\left(2^s d(b-1) + \begin{cases}
				2(2^s - 1) & 3 \mid b \\
				0 & \text{otherwise}
			\end{cases}\right) + \begin{cases}
				2 & 3 \mid b \\
				0 & \text{otherwise}
			\end{cases} \\
			&= 2^{s+1}d(b-1) + \begin{cases}
				2(2^{s+1} - 1) & 3 \mid b \\
				0 & \text{otherwise},
			\end{cases} 
		\end{align*}
		as desired.
	\end{proof}
	Thus, the values of $d(n)$ are determined by the values $d(b-1)$ for odd $b$, together with whether $b$ is divisible by 3.
	
	We can also use Theorem \ref{2nPlus1} to further restrict when $d(n) = 2$.
	\begin{corollary}\label{D2Minus1Mod6}
		If $d(n) = 2$, then $n \equiv -1 \pmod 6$.
	\end{corollary}
	\begin{proof}
		Since 2 is not divisible by 4, Corollary \ref{NEvenDDivisible4} tells us that $n$ must be odd.
		So, we can write $n = 2^k b - 1$ for $b$ odd and $k \geq 1$.
		Using Corollary \ref{2nPlus1Explicit}, we find
		\begin{equation*}
			d(2^k b - 1) = \begin{cases}
				2^k d(b-1) + 2(2^k - 1) & 3 \mid b \\
				2^k d(b-1) & \text{otherwise}
			\end{cases}.
		\end{equation*}
		If $3 \nmid b$, then we must have $d(n) = 2 = 2^k d(b-1)$.
		This is impossible because Corollary \ref{NEvenDDivisible4} tells us that $d(b-1)$ is divisible by 4.
		
		If $3 \mid b$, then we may write $b = 3j$.
		We also must have $d(n) = 2 = 2^k d(b-1) + 2(2^k - 1)$.
		If $k \geq 2$, then this expression is greater than 2.
		Thus, we must have $k=1$ and $2 = 2 d(b-1) + 2$, which simplifies to $d(b-1) = 0$.
		Solving for $n$, we find
		\begin{align*}
			n &= 2^k b - 1 \\
			&= 2b - 1 \\
			&= 6j - 1,
		\end{align*}
		so $n \equiv -1 \pmod 6$, as desired.
	\end{proof}

	The preceding results can help us find values of $d(n)$ from smaller known values, but they do not rule out any values of $d(n)$.
	To obtain such restrictions, we need to understand when individual polynomial factors can occur in the Fibonacci polynomial sequence.
	We introduce the notation of $\operatorname{rank}$ to record the first index at which a given polynomial appears as a factor.
	\begin{definition}
		Let $P$ be a non-zero polynomial in $\F_2[x]$.
		Then 
		\begin{equation*}
			\operatorname{rank}(P) = \min{\left\{r > 0 : P(x) \mid F_r(x)\right\}}.
		\end{equation*}
	\end{definition}
	Hunziker, Machiavelo, and Park proved that every polynomial divides some $F_r(x)$ and therefore has a finite $\operatorname{rank}$ \cite[Corollary 2.8]{HUNZIKER2004465}.
	
	We now show a relationship between the divisibility of polynomials and the divisibility of their ranks.
	\begin{lemma}\label{rankApparition}
		For any non-zero $P(x) \in \F_2[x]$,
		\begin{equation*}
			P(x) \mid F_m(x) \iff \operatorname{rank}(P) \mid m.
		\end{equation*}
	\end{lemma}
	\begin{proof}
		Let $r = \operatorname{rank}(P)$.
		
		Assume that $P(x)$ divides $F_m(x)$.
		$P(x)$ must also divide $F_r(x)$.
		So by Lemma \ref{FibonacciDivisibility},
		\begin{align*}
			P(x) &\mid \gcd{(F_r(x), F_m(x))} \\
			P(x) &\mid F_{\gcd(r, m)}(x).
		\end{align*}
		Since $r$ is minimal, $\gcd{(r,m)} = r$, and so $r \mid m$.
		
		Now assume that $r \mid m$.
		Then $\gcd{(r,m)} = r$.
		By Lemma \ref{FibonacciDivisibility}, $F_r(x) \mid F_m(x)$, and so $P(x) \mid F_m(x)$.
	\end{proof}
	
	Thus, the rank of a polynomial completely determines the indices at which it can divide a Fibonacci polynomial.
	We can now use the ranks to restrict the possible factors of the GCD defining $d(n)$, allowing us to rule out the smallest even value not attained by $d(n)$.
	\begin{theorem}\label{nodn26}
		The smallest even $k$ such that for all $n \in \N$, $d(n) \neq k$ is $k=26$.
	\end{theorem}
	\begin{proof}
		Recall from Lemma \ref{DEven} that $d(n)$ is always even.
		The following computations show that every positive even value less than 26 occurs:
		\begin{alignat*}{12}
			d(5) &= 2 \qquad & d(4) &= 4 \qquad & d(11) &= 6 \qquad & d(16) &= 8 \qquad & d(29) &= 10 \qquad & d(84) &= 12 \\
			d(23) &= 14 \qquad & d(19) &= 16 \qquad & d(101) &= 18 \qquad & d(30) &= 20 \qquad & d(59) &= 22 \qquad & d(62) &= 24.
		\end{alignat*}
		We will show that $d(n)=26$ is impossible.
		
		Assume for the sake of contradiction that there exists $n \in \N$ such that $d(n) = 26$.
		Since 26 is not divisible by 4, Corollary \ref{NEvenDDivisible4} says $n$ must be odd.
		So, we may write $n = 2m+1$.
		By Theorem \ref{2nPlus1},
		\begin{equation*}
			26 = d(n) = d(2m+1) = 2d(m) + \begin{cases}
				2 & 3 \mid m+1 \\
				0 & \text{otherwise}
			\end{cases}.
		\end{equation*}
		Since Lemma \ref{DEven} tells us that $d(m)=13$ is not possible, we must have $d(m) = 12$ and $m+1$ is divisible by 3.
		
		Let $m+1 = 2^s b$ for odd $b$ divisible by 3 and $s \geq 0$.
		By Corollary \ref{2nPlus1Explicit},
		\begin{align*}
			d(m) = 12 &= 2^s d(b-1) + 2(2^s - 1) \\
			14 &= 2^s(d(b-1) + 2) \\
			7 &= 2^{s-1}(d(b-1) + 2).
		\end{align*}
		The only possible integer solutions here are $d(b-1)=12$, $s=0$ or $d(b-1)=5$, $s=1$.
		Lemma \ref{DEven} tells us that $d(b-1) = 5$ is impossible, so the only remaining possibility is that $d(m) = d(b-1) = 12$ and $s=0$.
		Thus, $m$ must be even.
		
		Since $m+1$ is odd, Lemma \ref{SquareSquareFree} tells us that there exists some square-free $R_{m+1}(x)$ such that
		\begin{equation*}
			F_{m+1}(x) = R_{m+1}(x)^2.
		\end{equation*}
		Define $H_{m+1}(x)$ such that
		\begin{equation*}
			H_{m+1}(x) = \gcd{\left(R_{m+1}(x),R_{m+1}(x+1)\right)}.
		\end{equation*}
		Since squaring $R_{m+1}$ gives $F_{m+1}$,
		\begin{equation*}
			 H_{m+1}(x)^2 = \gcd{\left(F_{m+1}(x),F_{m+1}(x+1)\right)}.
		\end{equation*}
		Since $R_{m+1}(x)$ is square-free, $H_{m+1}(x)$ is too.
		Notice too that $H_{m+1}(x)$ is unaffected by translation: $H_{m+1}(x) = H_{m+1}(x+1)$.
		Since $d(m)$ is 12, $\deg H_{m+1}(x)$ must be 6.
		Lemma \ref{translationInvarianceField} tells us that for $y = x^2 + x$, there exists a square-free degree 3 polynomial $h_{m+1}(y) \in \F_2[y]$ such that
		\begin{equation*}
			H_{m+1}(x) = h_{m+1}(y).
		\end{equation*}
		Since $m+1$ is odd, we must also require that $h_{m+1}(0) = 1$.
		There are only three such polynomials in $\F_2[y]$ that satisfy all these requirements:
		\begin{equation*}
			C_1(y) = y^3 + 1 \qquad C_2(y) = y^3 + y + 1 \qquad C_3(y) = y^3 + y^2 + 1.
		\end{equation*}
		We will handle these case-by-case.
		
		Calculating the $\operatorname{rank}$ of our three candidates for $H_{m+1}$ with respect to $x$,
		% TODO: Actually show calculations for the ranks in an appendix
		\begin{alignat*}{3}
			\operatorname{rank}(C_1(y)) &= \operatorname{rank}{\left((x^2+x)^3 + 1\right)} &= 85 \\
			\operatorname{rank}(C_2(y)) &= \operatorname{rank}{\left((x^2+x)^3 + (x^2+x) + 1\right)} &= 63 \\
			\operatorname{rank}(C_3(y)) &= \operatorname{rank}{\left((x^2+x)^3 + (x^2+x)^2 + 1\right)} &= 63
		\end{alignat*}
		We'll show that none of these are viable candidates for $H_{m+1}$
	
		Assume for the sake of contradiction that $C_1 = H_{m+1}$.
		Then $C_1 \mid H_{m+1}$.
		Since $H_{m+1} \mid R_{m+1} \mid F_{m+1}$, then $C_1 \mid F_{m+1}$.
		Then by Lemma \ref{rankApparition}, $\operatorname{rank}(C_1) = 85 \mid m+1$.
		We also showed earlier that $3 \mid m+1$.
		So, $\operatorname{lcm}{(85,3)} = 255 \mid m+1$.
		Other polynomials with ranks dividing 255 must also divide $F_{m+1}$.
		Let
		\begin{equation*}
			P(x) = x^4 + x^3 + 1 \text{ and } Q(x) = x^4 + x^3 + x^2 + x + 1.
		\end{equation*}
		The ranks of $P(x)$ and $Q(x)$ are 15 and 17 respectively, which are both factors of 255.
		Thus, by Lemma \ref{rankApparition}, $P(x) \mid F_{m+1}(x+1)$.
		Substituting $x+1$ in place of $x$, we also obtain that $P(x+1) \mid F_{m+1}(x+1)$.
		Since $P(x+1) = Q(x)$, then $Q(x) \mid F_{m+1}(x)$.
		Substituting $x+1$ in place of $x$, we also obtain that $Q(x+1) \mid F_{m+1}(x+1)$.
		Since $P$ and $Q$ divide both $F_{m+1}(x)$ and $F_{m+1}(x+1)$, they both also divide the GCD, so $P$ and $Q$ divide $H_{m+1}^2$.
		Since $P$ and $Q$ are both square-free, $P$ and $Q$ divide $H_{m+1}$.
		Since $C_1$, $P$, and $Q$ are all pairwise coprime,
		\begin{align*}
			C_1 P Q &\mid H_{m+1} \\
			\deg{H_{m+1}} &\geq \deg{C_1} + \deg{P} + \deg{Q} = 14,
		\end{align*}
		which contradicts our assumption that the degree was 6.
		Thus $C_1$ is excluded as a candidate.
	
		Assume for the sake of contradiction that $C_2 = H_{m+1}$.
		Since $C_2 = H_{m+1}$, $C_2 \mid H_{m+1}$.
		Notice that $H_{m+1}$ must also divide $R_{m+1}(x)$ and $R_{m+1}(x+1)$.
		Similarly, $H_{m+1}$ divides both $F_{m+1}(x)$ and $F_{m+1}(x+1)$.
		Thus, $C_2 \mid F_{m+1}(x)$ and $F_{m+1}(x+1)$.
		By Lemma \ref{rankApparition}, $\operatorname{rank}(C_2) = 63 \mid m+1$.
		Since $C_3$ also has rank 63, Lemma \ref{rankApparition} tells us that $C_3 \mid F_{m+1}(x)$.
		Since $C_3$ is translation invariant, substituting $x+1$ for $x$ shows that $C_3$ also divides $F_{m+1}(x+1)$.
		Since $C_3$ divides both $F_{m+1}(x)$ and $F_{m+1}(x+1)$, then $C_3$ must also divide their GCD $H_{m+1}^2$.
		Then since $C_3$ is square-free, $C_3 \mid H_{m+1}$.
		Since $C_2$ and $C_3$ are coprime,
		\begin{align*}
			C_2 C_3 \mid H_{m+1} \\
			\deg{H_{m+1}} &\geq \deg{C_2} + \deg{C_3} = 12.
		\end{align*}
		which contradicts our assumption that the degree was 6.
		Thus $C_2$ is excluded as a candidate.
		We can apply the same argument with $C_2$ and $C_3$ swapped to show that $C_3$ may also be excluded as a candidate.
		
		We have shown that no such polynomial $H_{m+1}$ can exist.
		Therefore, there is no $3 \mid m+1$ with $d(m) = 12$ and hence no $n$ with $d(n) = 26$.
	\end{proof}

	Theorem \ref{nodn26} gives one impossible value of $d(n)$.
	The recurrence in Theorem \ref{2nPlus1Explicit} allows us to propagate this obstruction to infinitely many more values.
	\begin{corollary}
		Suppose $a \in \N$ is even and an impossible value of $d(n)$.
		Then $2a+2$ is also an impossible value of $d(n)$.
	\end{corollary}
	\begin{proof}
		Suppose for the sake of contradiction that there exists $n$ such that $d(n) = 2a+2$.
		Since $2a+2$ is not divisible by 4, then Corollary \ref{NEvenDDivisible4} tells us that $n$ must be odd, so we may write $n = 2m+1$.
		So, by Theorem \ref{2nPlus1},
		\begin{equation*}
			2a+2 = d(n) = d(2m+1) = 2d(m) + \delta_m \qquad \delta_m = \begin{cases}
				2 & 3 \mid m+1 \\
				0 & \text{otherwise}
			\end{cases}.
		\end{equation*}
		If $\delta_m = 0$, then $2a+2 = 2d(m)$.
		By Lemma \ref{DEven}, $d(m)$ is even, so $2d(m)$ is divisible by 4.
		This contradicts the fact that $2a+2$ is not divisible by 4 because $a$ is even.
		If $\delta_m = 2$, then $d(m)=a$, contradicting the impossibility of $a$ as an output of $d$.
	\end{proof}
	Expanding the recurrence, if $a$ is an impossible even value, then $2^t(a+2)-2$ is also impossible for all $t \geq 0$.
	In other words, $26, 54, 110, 222, 446, \ldots$ are all impossible values of $d(n)$.
	
	\section{Finite Fields}
	We now turn to finite field methods to study the Fibonacci polynomial GCDs that determine $d(n)$.
	In particular, we use properties of roots of unity and traces to analyze $d(p^k - 1)$ for prime $p$ and to characterize when $d(p-1) = 0$.
	
	\subsection{Orders of Prime Powers}
	\begin{lemma}\label{ordpj2}
		Let $p$ be an odd prime.
		Define $h = \ord[p]{2}$ and $c = v_p(2^h - 1)$, where
		$v_p(a)$ is the $p$-adic valuation of $a$.
		Then for every $j \geq 1$,
		\begin{equation*}
			\ord[p^j]{2} = h p^{\max{(0, j - c)}}.
		\end{equation*}
	\end{lemma}
	\begin{proof}
		Suppose $2^r \equiv 1 \pmod{p^j}$.
		Since $2^r \equiv 1 \pmod p$, the definition of $h$ gives $h \mid r$, so we may write $r = ht$.
		By the Lifting the Exponent Lemma \cite[Chapter 2 Section 2.7]{ireland_rosen_1990},
		\begin{align*}
			v_p(2^r - 1) &= v_p(2^{ht} - 1) \\
			&= v_p(2^h - 1) + v_p(t) \\
			&= c + v_p(t).
		\end{align*}
		Thus,
		\begin{equation*}
			p^j \mid 2^{ht} - 1 \iff c + v_p(t) \geq j.
		\end{equation*}
		The smallest possible integer $t$ satisfying this inequality is $t = p^{\max{(0, j-c)}}$, so
		\begin{equation*}
			\ord[p^j]{2} = h p^{\max{(0, j-c)}},
		\end{equation*}
		as desired.
	\end{proof}
	
	\begin{definition}
		A prime number $p$ is a \textit{Wieferich prime} when
		\begin{equation*}
			p^2 \mid 2^{p-1} - 1.
		\end{equation*}
	\end{definition}
	For an odd prime $p$, the Wieferich primes are precisely the primes for which $v_p(2^{\ord[p]{2}}-1) > 1$.
	Thus, they are exactly the exceptional primes for which the order formulation in Lemma \ref{ordpj2} does not take its simplest form.
	
	It is conjectured that infinitely many Wieferich primes exist, but only two are currently known: 1093 and 3511 \cite{CrandallSearch}.
	Computer searches have shown any more Wieferich primes must be greater than $2^{64}$ \cite{PrimeGridWieferich}.

	The following equivalent characterization of Wieferich primes will be useful because it expresses the Wieferich condition in terms of the multiplicative order appearing in Lemma \ref{ordpj2}.
	\begin{lemma}\label{wieferichEquivalence}
		An odd prime number $p$ is a Wieferich prime if and only if
		\begin{equation*}
			p^2 \mid 2^{\ord[p]{2}} - 1.
		\end{equation*}
	\end{lemma}
	\begin{proof}
		Let $n = \ord[p]{2}$.
		Then $p-1 = kn$ for some natural number $k$, and
		\begin{equation*}
			2^{p-1} - 1 = 2^{kn} - 1.
		\end{equation*}
		So, by the Lifting the Exponent lemma \cite[Chapter 2 Section 2.7]{ireland_rosen_1990},
		\begin{equation*}
			v_p(2^{kn} - 1) = v_p(2^n - 1) + v_p(k).
		\end{equation*}
		Since $1 \leq k \leq p-1$, we have $v_p(k) = 0$.
		Thus,
		\begin{equation*}
			v_p(2^{p-1} - 1) = v_p(2^n - 1).
		\end{equation*}
		So, $p^2 \mid 2^{p-1} - 1$ if and only if $p^2 \mid 2^{\ord[p]{2}} - 1$, as desired.
	\end{proof}
	Consequently, if $p$ is not a Wieferich prime, then $v_p(2^{\ord[p]{2}} - 1) = 1$.
	So, $c = 1$ and $\ord[p^j]{2} = \ord[p]{2} p^{j-1}$.

	\begin{corollary}\label{nonWieferichOrd}
		Let $p$ be an odd prime.
		If $p$ is not a Wieferich prime, then
		\begin{equation*}
			\ord[p^j]{2} = \ord[p]{2} p^{j-1}.
		\end{equation*}
	\end{corollary}
	\begin{proof}
		By Lemma \ref{ordpj2},
		\begin{equation*}
			\ord[p^j]{2} = \ord[p]{2} p^{\max{(0, j-c)}}, \qquad c = v_p(2^{\ord[p]{2}} - 1).
		\end{equation*}
		Thus it is sufficient to show that $c=1$.
		By definition of $\ord[p]{2}$, we have $c \geq 1$.
		Since $p$ is not a Wieferich prime, Lemma \ref{wieferichEquivalence} implies that
		\begin{equation*}
			p^2 \nmid 2^{\ord[p]{2}} - 1.
		\end{equation*}
		Thus $c < 2$ and therefore must be exactly 1, as desired.
	\end{proof}
	
	\subsection{Extension Degrees \& Traces}
	We next turn to finite field extensions and traces.
	These tools will allow us to relate the roots of Fibonacci polynomials to their behavior over finite fields.
	
	\begin{definition}
		Let $b$ be coprime to $n$.
		Then \textit{signed order} of $b$ modulo $n$ is
		\begin{equation*}
			\rho_b(n) = \min{\{r > 0 : b^r \equiv \pm 1 \pmod{n}\}}.
		\end{equation*}
	\end{definition}
	This is also sometimes called the \textit{index} of $"\pm 1$ relative to $b$, written $\operatorname{ind}_{b}(\pm 1)$.
	Note that compared to the typical order, the signed order is either the same value or half as much.
	
	\begin{lemma}\label{signedOrder}
		Let $p$ be an odd prime, $j \ge 1$, and $H_j = \ord[p^j]{2}$.
		Then
		\begin{equation*}
			\rho_{2}{(p^j)} = \begin{cases}
				H_j / 2 & H_j \text{ is even} \\
				H_j & H_j \text{ is odd}
			\end{cases}.
		\end{equation*}
		Moreover, when $H_j$ is even,
		\begin{equation*}
			2^{H_j / 2} \equiv -1 \pmod{p^j}.
		\end{equation*}
	\end{lemma}
	\begin{proof}
		Assume $H_j$ is even.
		Then by definition of $H_j$,
		\begin{equation*}
			\left(2^{H_j / 2}\right)^2 \equiv 1 \pmod{p^j}.
		\end{equation*}
		Thus the only solutions are $2^{H_j / 2} \equiv \pm 1 \pmod{p^j}$.
		However, 1 cannot be a solution; otherwise, $H_j$ would not be minimal.
		So,
		\begin{equation*}
			2^{H_j / 2} \equiv -1 \pmod{p^j}.
		\end{equation*}
		Notice that if $r \leq H_j / 2$, and $2^r \equiv -1 \pmod{p^j}$, then $2^{2r} \equiv 1 \pmod{p^j}$; since $H_j$ is minimal, $H_j \leq 2r$.
		Thus to satisfy both constraints we must have $H_j / 2 = r$.
		
		Now assume $H_j$ is odd.
		Notice that by definition, 
		\begin{equation*}
			H_j = \ord[p^j]{2} = \min{\{h > 0 : 2^h \equiv 1 \pmod{p^j}\}},
		\end{equation*} 
		so smaller solutions can only be of the form $2^r \equiv -1 \pmod{p^j}$.
		Then $2^{2r} \equiv 1 \pmod{p^j}$, which means $H_j \mid 2r$, and since $H_j$ is odd, $H_j \mid r$.
		Thus $H_j \leq r$, so $H_j$ is minimal, as desired.
	\end{proof}
	
	We can use the signed order to find the degree of certain extensions of $\F_2$.
	\begin{lemma}\label{fieldExtensionSize}
		Let $M > 1$ be odd; let $\zeta$ have exact order $M$; set $\alpha = \zeta + \zeta^{-1}$.
		Then
		\begin{equation*}
			\left[\F_2(\alpha) : \F_2\right] = \rho_2(M).
		\end{equation*}
	\end{lemma}
	\begin{proof}
		A standard finite field fact says
		\begin{equation*}
			\left[\F_2(\alpha) : \F_2\right] = \min{\{r > 0 : \alpha^{2^r} = \alpha\}}
		\end{equation*}
		\cite[Theorem 2.14, p.~52]{lidl-niederreiter}.
		By definition of $\alpha$, $\alpha^{2^r} = \zeta^{2^r} + \zeta^{-2^r}$, so $\alpha^{2^r} = \alpha$ is equivalent to $\zeta^{2^r} + \zeta^{-2^r} = \zeta + \zeta^{-1}$.
		
		For any non-zero $a,b \in \overline{\F_2}$, $a + a^{-1} = b + b^{-1} \iff (a+b)(ab+1) = 0$.
		So, either $a = b$ or $a = b^{-1}$.
		Taking $a = \zeta^{2^r}$ and $b = \zeta$, we obtain
		\begin{equation*}
			\zeta^{2^r} = \zeta \quad\text{ or }\quad \zeta^{2^r} = \zeta^{-1}.
		\end{equation*}
		Since $\zeta$ has exact order $M$, we can equivalently write
		\begin{equation*}
			2^{r} \equiv 1 \pmod{M} \quad\text{ or }\quad 2^r \equiv -1 \pmod{M}.
		\end{equation*}
		So, $\alpha^{2^r} = \alpha \iff 2^r \equiv \pm 1 \pmod{M}$, and
		\begin{equation*}
			\left[\F_2(\alpha) : \F_2\right] = \min{\{r > 0 : \alpha^{2^r} = \alpha\}} = \rho_2(M),
		\end{equation*}
		as desired.
	\end{proof}
	
	We will also need the trace of elements in finite field extensions.
	We first recall its standard expression in terms of the Frobenius automorphism \cite[Definition 2.22, p.~54]{lidl-niederreiter}.
	\begin{definition}[Lidl \& Niederreiter]\label{fieldTrace}
		Let $a$ be algebraic over $\F_q$.
		Then for every $z \in \F_q(a)$,
		\begin{equation*}
			\Tr[\F_q(a) / \F_q]{z} = \sum_{r=0}^{\left[\F_q(a) : \F_q\right] - 1}{z^{q^r}}.
		\end{equation*}
	\end{definition}

	\begin{corollary}\label{traceAt1}
		\begin{equation*}
			\Tr[\F_q(a) / \F_q]{1} = \left[\F_q(a) : \F_q\right].
		\end{equation*}
	\end{corollary}
	\begin{proof}
		Immediate from evaluating Definition \ref{fieldTrace} at $z=1$.
	\end{proof}
	
	\begin{lemma}\label{fieldTraceSumRoots}
		Let $M > 1$ be odd, and let $\zeta$ have exact order $M$.
		Let $\alpha = \zeta + \zeta^{-1}$.
		Define
		\begin{equation*}
			G_M = \{\pm 2^r \pmod{M} : r \geq 0\} \subseteq (\Z / M\Z)^{\times}.
		\end{equation*}
		Then
		\begin{equation*}
			\Tr[\F_2(\alpha) / \F_2]{\alpha} = \sum_{u \in G_M}{\zeta^{u}}.
		\end{equation*}
	\end{lemma}
	\begin{proof}
		Lemma \ref{fieldExtensionSize} showed that
		\begin{equation*}
			\left[\F_2(\alpha) : \F_2\right] = \rho_2(M).
		\end{equation*}
		So, by Definition \ref{fieldTrace},
		\begin{equation*}
			\Tr[\F_2(\alpha) / \F_2]{\alpha} = \sum_{r=0}^{\rho_2(M)-1}{\zeta^{2^r} + \zeta^{-2^r}}.
		\end{equation*}
		By the definition of $\rho_2(M)$, the residues $2^0, -2^0, 2^1, -2^1, \ldots, 2^{\rho_2(M)-1}, -2^{\rho_2(M)-1}$. are all distinct modulo $M$.
		Otherwise, we would obtain $2^r \equiv \pm 2^s \pmod M$, or equivalently, $2^{r-s} \equiv \pm 1 \pmod M$ for some $0 \leq s < r < \rho_2(M)$, which contradicts $\rho_2(M)$ being minimal.
	
		The residues are exactly the elements of $G_M$, so
		\begin{equation*}
			\Tr[\F_2(\alpha) / \F_2]{\alpha} = \sum_{u \in G_M}{\zeta^u},
		\end{equation*}
		as desired.
	\end{proof}

	\begin{corollary}\label{traceMobius}
		If $G_M = (\Z / M\Z)^{\times}$, then
		\begin{equation*}
			\Tr[\F_2(\alpha) / \F_2]{\alpha} = \mu(M) \pmod 2,
		\end{equation*}
		where $\mu$ is the number-theoretic Möbius function.
	\end{corollary}
	\begin{proof}
		From Lemma \ref{fieldTraceSumRoots}, we know that
		\begin{equation*}
			\Tr[\F_2(\alpha) / \F_2]{\alpha} = \sum_{u \in G_M}{\zeta^{u}}.
		\end{equation*}
		When $G_M = (\Z / M\Z)^{\times}$, we also have
		\begin{equation*}
			\Tr[\F_2(\alpha) / \F_2]{\alpha} = \sum_{0 \leq u < M, \gcd{(u,M)} = 1}{\zeta^{u}} = \mu(M) \pmod 2,
		\end{equation*}
		by Ramanujan's sum identity of roots of unity, as desired.
	\end{proof}
	
	\subsection{Applications to $d(n)$}
	We can now combine our previous results to greatly reduce the calculation of $d(p^k - 1)$ when $p$ is a prime number.
	\begin{theorem}[Prime Stabilization]\label{primePowerStabilization}
		Let $p$ be an odd prime and set $c = v_p(2^{\ord[p]{2}} - 1)$.
		Then for every $k \geq c$,
		\begin{equation*}
			d(p^k - 1) = d(p^c - 1).
		\end{equation*}
	\end{theorem}
	\begin{proof}
		Fix $k \geq c$.
		For $1 \leq j \leq k$, let
		\begin{equation*}
			R_j = \{\zeta + \zeta^{-1} : \ord{\zeta} = p^j\}.
		\end{equation*}
		By Lemma \ref{RootParameterization},
		\begin{equation*}
			\operatorname{roots}{F_{p^k}} = \bigcup_{j=1}^{k}{R_j}.
		\end{equation*}
	
		Suppose $\alpha, \alpha+1 \in \overline{\F_2}$ are both roots of $F_{p^k}$, with
		\begin{equation*}
			\alpha \in R_j, \qquad \alpha+1 \in R_\ell.
		\end{equation*}
		Because $\F_2(\alpha) = \F_2(\alpha+1)$, their extension degrees over $\F_2$ are equal.
		By Lemma \ref{fieldExtensionSize},
		\begin{align*}
			\left[\F_2(\alpha) : \F_2\right] &= \left[\F_2(\alpha+1) : \F_2\right] \\
			\rho_2(p^j) &= \rho_2(p^\ell) \\
			\rho_2(p) p^{\max{(0,j-c)}} &= \rho_2(p) p^{\max{(0, \ell-c)}}.
		\end{align*}
		Hence either $j,l \leq c$ or $j = \ell > c$.
		
		It remains to exclude the second possibility.
		Assume for the sake of contradiction that $j = \ell > c$.
		Then
		\begin{equation*}
			\alpha, \alpha+1 \in R_\ell = R_j.
		\end{equation*}
		Set
		\begin{equation*}
			M = p^j, \qquad m = p^{j-c},
		\end{equation*}
		and choose a primitive $M$th root $\zeta \in \overline{\F_2}$ such that
		\begin{equation*}
			\alpha = \zeta + \zeta^{-1}.
		\end{equation*}
		Since $\alpha+1 \in R_j$, there must exist an integer $s$ with $\gcd{(s,p)} = 1$ such that
		\begin{equation*}
			\alpha+1 = \zeta^{s} + \zeta^{-s}
		\end{equation*}
		Thus
		\begin{align*}
			1 + \alpha + \alpha+1 &= 0 \\
			1 + \zeta + \zeta^{-1} + \zeta^{s} + \zeta^{-s} &= 0,
		\end{align*}
		so $\zeta$ is a root of
		\begin{equation}\label{P_s(x)}
			P_s(x) = 1 + x + x^{M-1} + x^s + x^{M-s}.
		\end{equation}
		The element $\zeta^m$ has exact order $p^c$.
		Let $f(x)$ be the minimal polynomial of $\zeta^m$ over $\F_2$.
		Then
		\begin{equation*}
			\deg{f} = \ord[p^c]{2} = \ord[p]{2}.
		\end{equation*}
		Since $\zeta^m$ is a root of $f(x)$, $\zeta$ is a root of $f(x^m)$, and
		\begin{equation*}
			\deg{f(x^m)} = \ord[p]{2}m = \ord[p]{2}p^{j-c} = \ord[p^j]{2}.
		\end{equation*}
		This is exactly the degree of the minimal polynomial of $\zeta$, so $f(x^m)$ is that minimal polynomial, and
		\begin{equation*}
			f(x^m) \mid P_s(x).
		\end{equation*}
	
		Every polynomial has a unique decomposition by exponents modulo $m$:
		\begin{equation*}
			P_s(x) = \sum_{a=0}^{m-1}{x^a Q_a(x^m)}.
		\end{equation*}
		If $f(x^m) \mid P_s(x)$, uniqueness implies that $f(x^m)$ divides every $Q_a(x^m)$.
		However, only the constant exponent in \eqref{P_s(x)} is divisible by $m$.
		The other four exponents are congruent modulo $m$ to $\pm 1$ and $\pm s$ are are non-zero because $p \nmid s$.
		Therefore $Q_0(x^m) = 1$, which is constant.
		However, $f(x^m)$ is not constant and thus cannot divide $Q_0(x^m)$, a contradiction.
		
		Such a contradiction excludes every common root in $R_j$ for $j > c$.
		All common roots of $F_{p^k}(x)$ and $F_{p^k}(x+1)$ therefore lie in
		\begin{equation*}
			\bigcup_{j=1}^{c}{R_j} = \operatorname{roots}{F_{p^c}}.
		\end{equation*}
		Theorem \ref{nk-1>n-1} also tells us that such common roots are also common roots for larger powers of $p$.
		Lemma \ref{DEven} tells us that every root has multiplicity 2, so their GCDs have the same degree.
		Thus,
		\begin{equation*}
			d(p^k - 1) = d(p^c - 1),
		\end{equation*}
		as desired.
	\end{proof}

	The result simplifies even further when $p$ is not a Wieferich prime.
	\begin{corollary}\label{pk-1Equalsp-1}
		If $p$ is a non-Wieferich prime, then for every $k \geq 1$,
		\begin{equation*}
			d(p^k - 1) = d(p-1).
		\end{equation*}
	\end{corollary}
	\begin{proof}
		The case for $p=2$ is the known identity $d(2^k - 1) = 0$ \cite{Sutner1989}.
		For an odd non-Wieferich prime,
		\begin{equation*}
			c = v_p\left(2^{}-1\right) = 1,
		\end{equation*}
		so the desired result follows directly from Theorem \ref{primePowerStabilization}.
	\end{proof}

	Although Theorem \ref{primePowerStabilization} does not provide the tools to handle all Wieferich primes, it does provide the computational tools to show the result in Corollary \ref{pk-1Equalsp-1} also holds for the two known non-Wieferich primes.
	\begin{corollary}\label{pk-1Equalsp-1Wieferich}
		For all $k \geq 1$,
		\begin{equation*}
			d(1093^k - 1) = 0, \qquad d(3511^k - 1) = 0.
		\end{equation*}
	\end{corollary}
	\begin{proof}
		Calculating the relevant stabilization exponent for each prime,
		\begin{alignat*}{4}
			c_{1093} &= v_{1093}\left(2^{\ord[1093]{2}} - 1\right) &= v_{1093}\left(2^{364} - 1\right) &= 2 \\
			c_{3511} &= v_{3511}\left(2^{\ord[3511]{2}} - 1\right) &= v_{3511}\left(2^{1755} - 1\right) &= 2.
		\end{alignat*}
		Applying Theorem \ref{primePowerStabilization} and computing $d(p-1)$ and $d(p^2 - 1)$,
		\begin{alignat*}{3}
			d(1093^1 - 1) &= d(1093^2 - 1) &= 0 \\
			d(3511^1 - 1) &= d(3511^2 - 1) &= 0,
		\end{alignat*}
		and the desired result follows.
	\end{proof}

	Thus the result holds the same for every known prime, leading us to the following conjecture.
	\begin{conjecture}
		For all primes $p$ and all $k \geq 0$,
		\begin{equation*}
			d(p^k - 1) = d(p-1).
		\end{equation*}
	\end{conjecture}
	Given Corollaries \ref{pk-1Equalsp-1} and \ref{pk-1Equalsp-1Wieferich} and the currently known Wieferich primes, any counter-example would require a newly discovered Wieferich prime greater than $2^{64}$ \cite{PrimeGridWieferich}.
	
	We can combine this result with our recurrence for $d(2n + 1)$ to obtain an explicit formula for a broad family of indices.
	\begin{corollary}\label{d2p-1}
		Let $p$ be an odd non-Wieferich prime or Wieferich prime 1093 or 3511.
		Then for all $n,m \geq 0$,
		\begin{equation*}
			d(2^np^m - 1) = \begin{cases}
				0 & m = 0 \\
				2^n d(p-1) & m \geq 1, p \neq 3 \\
				2^{n+1} - 2 & m \geq 1, p  = 3
			\end{cases}.
		\end{equation*}
	\end{corollary}
	\begin{proof}
		The $m=0$ case reduces to what Sutner already showed \cite{Sutner1989}.
		For the other cases, we first apply Corollary \ref{2nPlus1Explicit} to obtain
		\begin{equation*}
			d(2^np^m - 1) = \begin{cases}
				0 & m = 0 \\
				2^n d(p^m-1) & m \geq 1, p \neq 3 \\
				2^{n+1} - 2 & m \geq 1, p  = 3
			\end{cases}.
		\end{equation*}
		We then apply Corollaries \ref{pk-1Equalsp-1} and \ref{pk-1Equalsp-1Wieferich} to change $d(p^m - 1)$ to $d(p-1)$.
	\end{proof}

	We can also use the trace results to identify primes for which $d(p-1) = 0$.
	\begin{theorem}\label{dp-1isZero}
		Let $p \equiv 3 \pmod 4$ be a prime where
		\begin{equation*}
			\left\{\pm 2^r \pmod p : r \geq 0\right\} = (\Z / p \Z)^{\times},
		\end{equation*}
		or equivalently,
		\begin{equation*}
			\ord[p]{2} \in \left\{p-1, \frac{p-1}{2}\right\}
		\end{equation*}
		Then $d(p - 1) = 0$.
	\end{theorem}
	\begin{proof}
		Assume for the sake of contradiction that $d(p-1) \neq 0$.
		Then $F_{p}(x)$ and $F_{p}(x+1)$ have a common factor and thus a common root $\alpha$, so
		\begin{equation*}
			F_{p}(\alpha) = F_{p}(\alpha+1) = 0.
		\end{equation*}
		By Lemma \ref{RootParameterization}, there must exist non-trivial $p$th roots of unity $\zeta$ and $\eta$ such that
		\begin{equation*}
			\alpha = \zeta + \zeta^{-1} \qquad \alpha+1 = \eta + \eta^{-1}.
		\end{equation*}
	
		We can see that $\F_2(\alpha+1) = \F_2(\alpha)$ ($\alpha$ and $\alpha+1$ can be obtained from each other by adding 1), and so by Corollary \ref{traceMobius}, 
		\begin{equation*}
			\Tr[\F_2(\alpha+1) / \F_2]{\alpha+1} = \Tr[\F_2(\alpha) / \F_2]{\alpha} = \mu(p) \pmod 2 = 1.
		\end{equation*}
	
		However, by Lemmas \ref{fieldExtensionSize} and \ref{signedOrder},
		\begin{equation*}
			[\F_2(\alpha) : \F_2] = [\F_2(\alpha+1) : \F_2] = \rho_2(p) = \frac{p-1}{2}.
		\end{equation*}
		Since $p \equiv 3 \pmod 4$, this value is odd.
		Applying the linearity of the trace and Corollary \ref{traceAt1},
		\begin{equation*}
			\Tr[\F_2(\alpha) / \F_2]{\alpha+1} = \Tr[\F_2(\alpha) / \F_2]{\alpha} + \Tr[\F_2(\alpha) / \F_2]{1} = \Tr[\F_2(\alpha) / \F_2]{\alpha} + 1 = 0.
		\end{equation*}
		Thus we have reached a contradiction.
	\end{proof}
	
	\subsubsection{Finite Support Stabilization}
	% TODO: The notation in the proofs in this section aren't consistent with the rest of the paper
	We can extend the results of Theorem \ref{primePowerStabilization} to find a stabilization result for any odd base, based on the unique prime factors of the base.
	
	\begin{definition}
		Let $S$ be a finite set of odd primes.
		For each $p \in S$, set
		\begin{equation}\label{finiteSupportVariables}
			h_p = \ord[p]{2}, \qquad c_p = v_p(2^{h_p}-1), \qquad a_p(S) = \max_{q\in S}{v_p(h_q)}, \qquad C_p(S) = c_p + a_p(S).
		\end{equation}
		Let $N$ be an odd integer supported on $S$.
		Then the ``$S$-truncation'' of $N$ is
		\begin{equation}\label{S-truncation}
			N^\flat = \prod_{p \in S}{p^{\min{\left(v_p(N), C_p(S)\right)}}}.
		\end{equation}
	\end{definition}
	
	\begin{lemma}\label{orderGrowthTruncation}
		Let $Q$ be an odd integer supported on $S$, and let $Q^\flat$ be its $S$-truncation.
		Set
		\begin{equation*}
			m = \frac{Q}{Q^\flat}.
		\end{equation*}
		Then
		\begin{equation*}
			\ord[Q]{2} = m \ord[Q^\flat]{2}.
		\end{equation*}
	\end{lemma}
	\begin{proof}
		Write $u_p = v_p(Q)$.
		The Chinese Remainder Theorem and Theorem \ref{primePowerStabilization} give
		\begin{equation}\label{eq:order-lcm}
			\ord[Q]{2} = \mathop{\operatorname{lcm}}_{\substack{p\in S\\ u_p>0}}
			\left(h_p\,p^{\max(0,u_p-c_p)}\right)
		\end{equation}
		
		First consider a prime $\ell \notin S$.
		Its valuation in \eqref{eq:order-lcm} comes only from the fixed integers $h_p$.
		Truncation changes no prime support, so
		\begin{equation*}
			v_\ell(\ord[Q]{2}) = v_\ell(\ord[Q^\flat]{2}).
		\end{equation*}
	
		Now fix $p \in S$.
		The $p$-adic validation in \eqref{eq:order-lcm} is the maximum of
		\begin{equation*}
			\max{(0, u_p - c_p)} \qquad \text{and} \qquad \{v_p(h_q) : q \in S, q\neq p, u_q > 0\}.
		\end{equation*}
		Every term in the second set is at most $a_p(S)$.
		If $u_p \leq C_p(S)$, truncation leaves $u_p$ unchanged, and truncating the other exponents does not change their fixed factors $h_q$.
		The $p$-adic valuation of the order is therefore unchanged.
		
		If $u_p > C_p(S) = c_p + a_p(S)$, then
		\begin{equation*}
			u_p - c_p > a_p(S),
		\end{equation*}
		so the contribution from the $p$-power component strictly dominates all cross-prime contributions.
		Hence
		\begin{equation*}
			v_p(\ord[Q]{2}) = u_p - c_p.
		\end{equation*}
		At the truncated exponent $C_p(S)$, the corresponding valuation is $a_p(S)$, and therefore
		\begin{equation*}
			v_p(\ord[Q]{2}) - v_p(\ord[Q^\flat]{2}) = u_p - C_p(S) = v_p(m).
		\end{equation*}
		Comparing valuations at every prime proves the result.
	\end{proof}
	
	\begin{theorem}[Finite Support Truncation]\label{finiteSupportTruncation}
		Let $S$ be a finite set of odd primes and let $N$ be an odd positive integer supported on $S$.
		Then
		\begin{equation*}
			d(N - 1) = d(N^\flat - 1),
		\end{equation*}
		where $N^\flat$ is defined by \eqref{S-truncation}.
	\end{theorem}
	\begin{proof}
		Support $\alpha, \alpha+1 \in \overline{\F_2}$ are both roots of $F_N$.
		By Lemma \ref{RootParameterization}, there are divisors $M,L > 1$ of $N$ such that
		\begin{equation*}
			\alpha \in R_M, \qquad \alpha+1 \in R_L.
		\end{equation*}
		Because $\F_2(\alpha) = \F_2(\alpha+1)$, Lemma \ref{fieldExtensionSize} gives
		\begin{equation*}
			\rho_2(M) = \rho_2(L).
		\end{equation*}
		Write $j_p = v_p(M)$ and $\ell_p = v_p(L)$.
		Fix $p \in S$ and suppose $j_p > C_p(S)$.
		As in the proof of Lemma \ref{orderGrowthTruncation}, the $p$-part belonging to $p^{j_p}$ dominates every cross-prime contribution, so
		\begin{equation*}
			v_p(\ord[M]{2}) = j_p - c_p > a_p(S).
		\end{equation*}
	
		Signed order differs from ordinary order by at most a factor of two, so their validations at odd prime $p$ agree.
		Therefore,
		\begin{equation*}
			v_p(\ord[M]{2}) = j_p - c_p.
		\end{equation*}
		This valuation is larger than $a_p(s)$, so it cannot arise from any cross-prime order $h_p$.
		Consequently $\ell_p > C_p(S)$, and
		\begin{equation*}
			\ell_p - c_p = j_p - c_p, \qquad \text{hence} \qquad \ell_p = j_p.
		\end{equation*}
		By symmetry, every exponent above its threshold occurs with the same value in $M$ and $L$.
		Set
		\begin{equation*}
			Q = \operatorname{lcm}(M,L), \qquad Q^\flat = \prod_{p \in S}{p^{\min{(v_p(Q),C_p(S)})}}, \qquad m = \frac{Q}{Q^\flat}.
		\end{equation*}
	
		If $m=1$, then $M,L \mid N^\flat$ and there is nothing to prove.
		Suppose $m > 1$.
		Choose a primitive $Q$th root $\theta$.
		There are exponents $a,b$ such that
		\begin{equation*}
			\alpha = \theta^{a} + \theta^{-a}, \qquad \alpha+1 = \theta^{b}+\theta^{-b}.
		\end{equation*}
		where $\theta^a$ and $\theta^b$ have exact orders $M$ and $L$ respectively.
		Thus $\theta$ is a root of
		\begin{equation}\label{thetaRoot}
			P(x) = 1 + x^a + x^{Q-a} + x^b + x^{Q-b}.
		\end{equation}
	
		Every prime dividing $m$ occurs above its threshold in $Q$.
		The preceding rigidity argument say that its full exponent in $Q$ occurs in both $M$ and $L$.
		Since $\theta^a$ has exact order $M$, we have
		\begin{equation*}
			\gcd{(Q,a)} = \frac{Q}{M}.
		\end{equation*}
		No prime dividing $m$ divides $Q/M$, and no such prime can divide the remaining unit factor in $a$.
		Hence $\gcd{(b,m)} = 1$.
		Because $m \mid Q$, only the constant exponent in \eqref{thetaRoot} is divisible by $m$.
	
		The element $\theta^m$ has exact order $Q^\flat$.
		Let $f(x)$ be its minimal polynomial over $\F_2$.
		Lemma \ref{orderGrowthTruncation} gives
		\begin{equation*}
			\deg{f(x^m)} = m \ord[Q^\flat]{2} = \ord[Q]{2},
		\end{equation*}
		the degree of the minimal polynomial of $\theta$.
		Since $\theta$ is a root of $f(x^m)$, the polynomial $f(x^m)$ is its minimal polynomial.
		Therefore $f(x^m) \mid P(x)$.
		Decompose \eqref{thetaRoot} uniquely by its exponents modulo $m$:
		\begin{equation*}
			P(x) = \sum_{r=0}^{m-1}{x^r P_r(x^m)}.
		\end{equation*}
		Divisibility by $f(x^m)$ implies that $f(y)$ divides every $P_r(y)$.
		By the preceding coprimality shows that $P_0(y)=1$, a contradiction.
		Thus $m=1$ and every common root of $F_N(x)$ and $F_N(x+1)$ already occurs for $F_{N^\flat}(x)$ and $F_{N^\flat}(x+1)$.
		The converse include follows from $N^\flat \mid N$ and Theorem \ref{nk-1>n-1}.
		Because both indices are odd, every root has multiplicity two.
		Therefore,
		\begin{equation*}
			d(N-1) = d(N^\flat - 1),
		\end{equation*}
		as desired.
	\end{proof}

	\begin{corollary}[Eventual Consistency]
		For every odd positive integer $A$, there exists $k_0$ such that for all $k \geq k_0$,
		\begin{equation*}
			d(A^k - 1) = d(A^{k_0} - 1).
		\end{equation*}
		In particular,
		\begin{equation*}
			k_0 = \max_{p \in S}{\lceil\frac{C_p(S)}{b_p}\rceil}.
		\end{equation*}
	\end{corollary}
	\begin{proof}
		For every $k \geq k_0$, the truncation of $A^k$ is the fixed integer
		\begin{equation*}
			B_S = \prod_{p \in S}{p^{C_p(S)}}.
		\end{equation*}
		Theorem \ref{finiteSupportTruncation} gives
		\begin{equation*}
			d(A^k - 1) = d(B_S - 1).
		\end{equation*}
		The same theorem applied to $A^{k_0}$ gives
		\begin{equation*}
			d(A^{k_0} - 1) = d(B_S - 1),
		\end{equation*}
		which proves the claim.
	\end{proof}
